# Hash-locked pip for CI. Every workflow that upgrades the runner's pip installs
# this file with the command below, so the download is checked against the
# hashes below:
#   python -m pip install --require-hashes -r scripts/ci/requirements-ci-pip.txt
#
# pip checks hashes only when it downloads: a runner that already
# has this version reports "Requirement already satisfied" and verifies nothing.
#
# Floor: pip 26.2 or later fixes CVE-2026-13346 (earlier pip mishandles
# doubly-encoded package URLs from indexes). scripts/dev/check_pip_pin_drift.py
# fails the build when this file pins a pip below that floor.
#
# Dependabot (the pip entry for /scripts/ci in .github/dependabot.yml) proposes
# upgrades and rewrites the hashes. To regenerate by hand, put the one-line
# input `pip==<version>` in a scratch file, run the command below, and paste its
# output beneath this comment block (the hashes cover the wheel and the sdist):
#   uv pip compile <scratch>/in.txt --generate-hashes --no-header --no-annotate -o <scratch>/out.txt
pip==26.2.2 \
    --hash=sha256:3a8e5c800492787f33a4114ef54444546c340df428974ebba17e0fd47e11af2f \
    --hash=sha256:fd7f4dfa8190e014013b9f08f3e973f4e4c26ffd179a204895533a7254813a70
